cmake_minimum_required(VERSION 3.22)
project(citron_beta54_3_regressions LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_path(FMT_INCLUDE_DIR fmt/format.h REQUIRED)
option(BETA54_3_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
find_path(NLOHMANN_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
enable_testing()
foreach(name ctest-response domain-object-table member-label last-nat-cache native-friend-code bcat-cache-tools applet-touch buffer-cache-address-space viewer-presence setup-startup home-menu-metadata external-application-metadata)
    add_executable(${name} ${name}.cpp)
    target_include_directories(${name} PRIVATE ../../src ${NLOHMANN_JSON_INCLUDE_DIR} ${FMT_INCLUDE_DIR})
    target_link_libraries(${name} PRIVATE Threads::Threads)
    if(BETA54_3_SANITIZERS)
        if(MSVC)
            message(FATAL_ERROR "Use GCC or Clang for the ASan/UBSan checks")
        endif()
        target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${name} PRIVATE -fsanitize=address,undefined)
    endif()
    # These checks use assertions; never silently compile them out in Release.
    if(MSVC)
        target_compile_options(${name} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${name} PRIVATE -UNDEBUG)
    endif()
    if(name STREQUAL "last-nat-cache")
        add_test(NAME ${name} COMMAND ${name} ${CMAKE_CURRENT_BINARY_DIR})
    else()
        add_test(NAME ${name} COMMAND ${name})
    endif()
endforeach()

find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(profile_test_flags)
if(BETA54_3_SANITIZERS)
    list(APPEND profile_test_flags --sanitizers)
endif()
add_test(NAME profile-ipc-latency COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/profile-ipc-latency.py
    --compiler ${CMAKE_CXX_COMPILER} ${profile_test_flags})

find_package(Qt6 COMPONENTS Core QUIET)
if(TARGET Qt6::Core)
    add_executable(temporary-directory temporary-directory.cpp)
    target_include_directories(temporary-directory PRIVATE ../../src)
    target_link_libraries(temporary-directory PRIVATE Qt6::Core)
    if(MSVC)
        target_compile_options(temporary-directory PRIVATE /UNDEBUG)
    else()
        target_compile_options(temporary-directory PRIVATE -UNDEBUG)
    endif()
    if(BETA54_3_SANITIZERS)
        target_compile_options(temporary-directory PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(temporary-directory PRIVATE -fsanitize=address,undefined)
    endif()
    if(WIN32)
        add_custom_command(TARGET temporary-directory POST_BUILD COMMAND ${CMAKE_COMMAND}
            -E copy_if_different $<TARGET_FILE:Qt6::Core> $<TARGET_FILE_DIR:temporary-directory>)
    endif()
    add_test(NAME temporary-directory COMMAND temporary-directory)
endif()
